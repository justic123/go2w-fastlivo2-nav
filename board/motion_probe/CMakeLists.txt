cmake_minimum_required(VERSION 3.16)
project(go2w_motion_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(unitree_sdk2 REQUIRED)
find_path(GO2W_JSON_INCLUDE json.hpp HINTS /unitree/module/unitree_slam/example/include)
if(NOT GO2W_JSON_INCLUDE)
  message(FATAL_ERROR "The existing json.hpp dependency is required")
endif()
add_executable(go2w_motion_probe motion_probe.cpp)
target_include_directories(go2w_motion_probe PRIVATE ${GO2W_JSON_INCLUDE})
target_link_libraries(go2w_motion_probe unitree_sdk2)
